# STM32 通用定时器中断

```c
#include "TIMER/timer.h"
#include "stm32f10x.h"     //必须添加的头文件
#include "stm32f10x_tim.h" //必须添加的头文件

/**
 * @brief  TIM3 定时器初始化函数
 * @param  arr: 自动重装载值 (0~65535)
 * @param  psc: 预分频器值 (0~65535)
 * @retval 无
 * @note   中断时间计算公式: T = (arr + 1) * (psc + 1) / 定时器时钟频率
 *         例: TIM3时钟=72MHz, arr=999, psc=7199 → T=1000*7200/72MHz=0.1s=100ms
 */
void TIM3_Int_Init(u16 arr, u16 psc)
{   
    NVIC_InitTypeDef NVIC_InitStructure;           // 定义NVIC结构体变量
    TIM_TimeBaseInitTypeDef TIM_TimeBaseInitStructure;  // 定义定时器结构体变量
    
    // 1. 使能TIM3时钟
    // TIM3挂在APB1总线上，需要使能APB1时钟
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM3, ENABLE);
    
    // 2. 配置定时器基础参数
    TIM_TimeBaseInitStructure.TIM_Period = arr;      // 自动重装载值(计数器从0数到此值产生中断)
    TIM_TimeBaseInitStructure.TIM_Prescaler = psc;   // 预分频器(对定时器时钟进行分频)
    TIM_TimeBaseInitStructure.TIM_ClockDivision = 0; // 时钟分频因子(一般设为0)
    TIM_TimeBaseInitStructure.TIM_CounterMode = TIM_CounterMode_Up;  // 向上计数模式
    
    // 3. 初始化TIM3
    TIM_TimeBaseInit(TIM3, &TIM_TimeBaseInitStructure);
    
    // 4. 使能TIM3的更新中断
    // TIM_IT_Update: 更新中断，当计数器溢出时触发
    TIM_ITConfig(TIM3, TIM_IT_Update, ENABLE);
    
    // 5. 配置NVIC中断优先级
    NVIC_InitStructure.NVIC_IRQChannel = TIM3_IRQn;           // 中断通道:TIM3
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0; // 抢占优先级(0最高)
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 3;        // 子优先级(0最高)
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;           // 使能中断通道
    NVIC_Init(&NVIC_InitStructure);                           // 初始化NVIC
    
    // 6. 使能TIM3计数器(开始计时)
    TIM_Cmd(TIM3, ENABLE);
}

/**
 * @brief  TIM3 中断服务函数
 * @note   当定时器计数器溢出时自动调用此函数
 *         函数名 TIM3_IRQHandler 是固定的，由启动文件决定
 */
void TIM3_IRQHandler(void)
{
    // 检查TIM3的更新中断标志位是否被置位
    // TIM_IT_Update: 更新中断标志，计数器溢出时硬件置1
    if (TIM_GetITStatus(TIM3, TIM_IT_Update) != RESET)
    {
        LED1 = !LED1;  // 翻转LED1状态(亮变灭或灭变亮)
        
        // 清除中断标志位(必须操作，否则会一直进入中断)
        TIM_ClearITPendingBit(TIM3, TIM_IT_Update);
    }
}
```

## 常用参数速查表

| 目标中断时间    | arr  | psc  | 计数器时钟 | 说明     |
| --------- | ---- | ---- | ----- | ------ |
| **1ms**   | 999  | 71   | 1MHz  | 常用任务调度 |
| **10ms**  | 9999 | 71   | 1MHz  | 10ms周期 |
| **100ms** | 999  | 7199 | 10kHz | 简单延时   |
| **500ms** | 4999 | 7199 | 10kHz | 慢闪LED  |
| **1秒**    | 9999 | 7199 | 10kHz | 1秒周期   |

```c
1. 预分频后的计数器时钟频率 = 定时器时钟 / (psc + 1)
2. 计数器从 0 计数到 arr 产生一次中断，需要 (arr + 1) 个时钟周期
3. 中断时间 = (arr + 1) / 计数器时钟频率
           = (arr + 1) × (psc + 1) / 定时器时钟
```
